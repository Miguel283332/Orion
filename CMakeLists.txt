cmake_minimum_required(VERSION 3.16)
project(Demostrador VERSION 4.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Tipo de build" FORCE)
endif()

option(DEMO_BUILD_TESTS "Compilar los tests" ON)
option(DEMO_WARNINGS_AS_ERRORS "Tratar warnings como errores" OFF)

# nlohmann/json: del sistema o descargado al configurar
find_package(nlohmann_json 3.9 QUIET)
if(NOT nlohmann_json_FOUND)
  message(STATUS "nlohmann_json no encontrado: se descargará (FetchContent)")
  include(FetchContent)
  FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz)
  FetchContent_MakeAvailable(json)
endif()
find_package(Threads REQUIRED)

add_library(demo_core STATIC src/util.cpp src/llm.cpp src/verifier.cpp src/engine.cpp)
target_include_directories(demo_core PUBLIC include)
target_link_libraries(demo_core PUBLIC nlohmann_json::nlohmann_json Threads::Threads)
if(WIN32)
  target_compile_definitions(demo_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(demo_core PUBLIC -Wall -Wextra -Wno-unused-parameter)
  if(DEMO_WARNINGS_AS_ERRORS)
    target_compile_options(demo_core PUBLIC -Werror)
  endif()
elseif(MSVC)
  # la optimización la decide el tipo de build (no forzar /O2: choca con /RTC1 de Debug)
  target_compile_options(demo_core PUBLIC /W3 /utf-8)
endif()

add_executable(demostrador src/main.cpp)
target_link_libraries(demostrador PRIVATE demo_core)
target_compile_definitions(demostrador PRIVATE DEMO_VERSION="${PROJECT_VERSION}")

if(DEMO_BUILD_TESTS)
  enable_testing()
  add_executable(demo_tests tests/test_main.cpp)
  target_link_libraries(demo_tests PRIVATE demo_core)
  target_compile_definitions(demo_tests PRIVATE DEMO_TESTS_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests")
  add_test(NAME demo_tests COMMAND demo_tests)
endif()

install(TARGETS demostrador RUNTIME DESTINATION bin)
install(DIRECTORY problemas/ DESTINATION share/demostrador/problemas)
message(STATUS "Demostrador ${PROJECT_VERSION}: build=${CMAKE_BUILD_TYPE}")
